Quick Snippet: SQLite Ingestion Function 
import sqlite3
import pandas as pd

def ingest_medical_data(csv_path, db_name="revenue.db"):
    conn = sqlite3.connect(db_name)
    df = pd.read_csv(csv_path)
    df.to_sql("medical_records", conn, if_exists="append", index=False)
    conn.close()
    print("Ingestion complete!")
